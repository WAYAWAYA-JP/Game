# 設計書：マップ横拡張 & 難易度配置

**作成日**: 2026-03-20  
**依存**: `scripts/entities/map_data.gd` / `scripts/windows/map_display.gd` / `scripts/windows/map_window.gd`  
**方針**: Kingswayを意識した横長マップ。序盤→弱敵、中盤→中級敵、終盤→強敵の配置。スクロールして探索するイメージ。

---

## 概要

現状のマップ（10〜14列）を **約3倍の30〜42列** に拡張する。  
プレイヤーはマップウィンドウを横スクロールしながら進行する。  
ノードの数が増える分、敵の難易度を **進行度（カラム位置のtier）** に基づいて制御する。

```
[スタート] → [序盤ゾーン 弱敵] → [中盤ゾーン 中級敵] → [ミニボス] → [終盤ゾーン 強敵] → [ボス]
  col.0      col.1〜col.10      col.11〜col.25      col.26前後    col.27〜col.38     col.最終
(progress:0) (progress: 〜0.33) (progress: 〜0.70)               (progress: 〜0.95)
```

---

## Kingsway参考ポイント

| Kingsway要素 | 今回の対応 |
|-------------|-----------|
| 横に長いマップをスクロールして進む | MapDisplayを横長キャンバスに、ScrollContainerで包む |
| 序盤は弱い敵から始まる | tier 1の敵のみ出現（progress < 0.33） |
| 中盤でバリエーションが増える | tier 1〜2の敵が混在（progress 0.33〜0.70） |
| 終盤は強敵だらけ | tier 2〜3の敵が出現（progress > 0.70） |
| ミニボス的な節目がある | MINIBOSSノードを中盤後半に配置 |

**`_pick_column_type()` の難易度ゾーン分け更新**

```gdscript
func _pick_column_type(column: int) -> int:
    var progress: float = float(column) / float(_num_columns - 1)
    var roll: float = randf()

    # 中盤ミニボス列（progress 約0.58付近の1列）
    if column == int(_num_columns * 0.58):
        return TileType.MINIBOSS

    # ボス直前の回復地点（ボス3列前）
    if column == _num_columns - 3:
        if roll < 0.40: return TileType.TOWN
        elif roll < 0.70: return TileType.SHOP
        elif roll < 0.88: return TileType.EVENT
        else: return TileType.DUNGEON

    # ボス直前列
    if column == _num_columns - 2:
        if roll < 0.50: return TileType.DUNGEON
        elif roll < 0.70: return TileType.SHOP
        else: return TileType.EVENT

    # 序盤 progress < 0.33
    if progress < 0.33:
        if roll < 0.20: return TileType.TOWN
        if roll < 0.40: return TileType.SHOP
        if roll < 0.65: return TileType.EVENT
        return TileType.EMPTY

    # 中盤 0.33 <= progress < 0.70
    if progress < 0.70:
        if roll < 0.05: return TileType.TOWN
        if roll < 0.15: return TileType.SHOP
        if roll < 0.45: return TileType.DUNGEON
        if roll < 0.65: return TileType.EVENT
        return TileType.EMPTY

    # 終盤 progress >= 0.70
    if roll < 0.55: return TileType.DUNGEON
    if roll < 0.70: return TileType.EVENT
    if roll < 0.80: return TileType.SHOP
    return TileType.EMPTY
```

func _spawn_miniboss_battle() -> void:
    if _enemy_database.is_empty():
        return
    # DB全体の70%付近のエントリを中ボスとして使用
    var mid_idx: int = int(_enemy_database.size() * 0.70)
    var enemy_data: Dictionary = _enemy_database[mini(mid_idx, _enemy_database.size()-1)].duplicate(true)
    enemy_data["hp"] = int(enemy_data["hp"] * 1.3)
    enemy_data["attack"] = int(enemy_data["attack"] * 1.3)
    enemy_data["reward_gold"] = int(enemy_data["reward_gold"] * 2.0)
    enemy_data["reward_exp"] = int(enemy_data["reward_exp"] * 2.0)
    enemy_data["name"] = "中ボス: " + enemy_data["name"]
    WindowManager.create_window("battle", {
        "position": Vector2(200 + randf() * 100, 60 + randf() * 60),
        "enemy_data": enemy_data
    })
    info_label.text = "ミニボスバトル！"
    if not EventBus.battle_ended.is_connected(_on_miniboss_battle_ended):
        EventBus.battle_ended.connect(_on_miniboss_battle_ended, CONNECT_ONE_SHOT)

func _on_miniboss_battle_ended(victory: bool) -> void:
    if victory:
        _find_random_item()
        _find_random_item()
        var gold: int = 150 + randi() % 101
        GameState.add_gold(gold)
        event_log.text = "中ボスを撃破！ アイテム×2 と %dG を入手した！" % gold
    else:
        event_log.text = "中ボスに敗れた…"
    info_label.text = "接続先のノードをクリックして移動"

**`_spawn_battle()` を進行度ベースの敵選択に変更**

```gdscript
func _spawn_battle(use_normal_callback: bool = true) -> void:
    if _enemy_database.is_empty():
        event_log.text = "影が通り過ぎた...（敵データなし）"
        return
    var enemy_data: Dictionary = _pick_enemy_by_progress()
    # フロアスケーリングは既存処理のまま
    if map_data.floor_number > 1:
        var scale: float = 1.0 + (map_data.floor_number - 1) * 0.3
        enemy_data["hp"] = int(enemy_data["hp"] * scale)
        enemy_data["attack"] = int(enemy_data["attack"] * scale)
        enemy_data["reward_gold"] = int(enemy_data["reward_gold"] * scale)
        enemy_data["reward_exp"] = int(enemy_data["reward_exp"] * scale)
    WindowManager.create_window("battle", {...})
    ...

func _pick_enemy_by_progress() -> Dictionary:
    var tier: int = _progress_to_tier(_get_current_progress())
    var pool: Array = _enemy_database.filter(func(e): return e.get("tier", 1) == tier)
    if pool.is_empty():
        return _enemy_database[randi() % _enemy_database.size()].duplicate(true)
    return pool[randi() % pool.size()].duplicate(true)

func _get_current_progress() -> float:
    var node: Dictionary = map_data.get_node_by_id(map_data.current_node_id)
    if node.is_empty():
        return 0.0
    return float(node.get("column", 0)) / float(map_data._num_columns - 1)

func _progress_to_tier(progress: float) -> int:
    if progress < 0.33: return 1
    elif progress < 0.70: return 2
    else: return 3
```
